\section{Logic \& Equivalences}
\textbf{Operators:} $\neg$ (not), $\land$ (and), $\lor$ (or), $\oplus$ (xor), $\rightarrow$ (if), $\leftrightarrow$ (iff). \\
\textbf{Order:} $\neg, \land, \lor, \rightarrow, \leftrightarrow$ (High to Low). \\
\textbf{Equivalences:}
\begin{itemize}
    \item $p \rightarrow q \equiv \neg p \lor q$
    \item $\neg(p \land q) \equiv \neg p \lor \neg q$ (De Morgan)
    \item $p \leftrightarrow q \equiv (\neg p \lor q) \land (\neg q \lor p)$
    \item \textbf{Identity:} $p \land T \equiv p$; $p \lor F \equiv p$
    \item \textbf{Domination:} $p \lor T \equiv T$; $p \land F \equiv F$
    \item \textbf{Absorption:} $p \lor (p \land q) \equiv p$; $p \land (p \lor q) \equiv p$
\end{itemize}

\textbf{Fallacies:}
\begin{itemize}
    \item \textbf{Affirming Consequent:} $(p \rightarrow q) \land q \therefore p$ (Error)
    \item \textbf{Denying Antecedent:} $(p \rightarrow q) \land \neg p \therefore \neg q$ (Error)
\end{itemize}

\section{Rules of Inference}
\begin{tabular}{ll}
    $p, p \rightarrow q \therefore q$ & Modus Ponens \\
    $\neg q, p \rightarrow q \therefore \neg p$ & Modus Tollens \\
    $p \rightarrow q, q \rightarrow r \therefore p \rightarrow r$ & Hypo. Syllogism \\
    $p \lor q, \neg p \therefore q$ & Disj. Syllogism \\
    $p \therefore p \lor q$ & Addition \\
    $p \land q \therefore p$ & Simplification \\
    $p, q \therefore p \land q$ & Conjunction \\
    $p \lor q, \neg p \lor r \therefore q \lor r$ & Resolution
\end{tabular}

\section{Predicate Logic}
\textbf{Universal:} $\forall x P(x)$ (True for all $x$ in domain) \\
\textbf{Existential:} $\exists x P(x)$ (True for at least one $x$) \\
\textbf{Negation:} \\
$\neg \forall x P(x) \equiv \exists x \neg P(x)$ \\
$\neg \exists x P(x) \equiv \forall x \neg P(x)$

% =========================
\section{Proof Techniques}
\textbf{Direct:} Assume $p$, show $q$. \\
\textbf{Contrapositive:} Assume $\neg q$, show $\neg p$. \\
\textbf{Contradiction:} Assume $p \land \neg q$, derive $F$. \\
\textbf{Cases:} Prove $(p_1 \lor p_2 \lor \dots \lor p_n) \rightarrow q$ by proving $p_i \rightarrow q$ for each $i$.

\section{Sets \& Cardinality}
$A \subseteq B \iff \forall x(x \in A \rightarrow x \in B)$ \\
\textbf{Operations:} $A \cup B$ (or), $A \cap B$ (and), $A - B$ ($A$ not in $B$). \\
\textbf{Complement:} $\overline{A} = U - A$. \\
\textbf{Power Set:} $P(A) = \{S \mid S \subseteq A\}$, $|P(A)| = 2^{|A|}$. \\
\textbf{Cartesian:} $A \times B = \{(a,b) \mid a \in A, b \in B\}$, $|A \times B| = |A||B|$. \\
\textbf{Inclusion-Exclusion:} $|A \cup B| = |A| + |B| - |A \cap B|$.

% =========================
\section{Functions}
\textbf{Function:} $f: A \to B$. $A$ is domain, $B$ is codomain. \\
\textbf{Injective (1-1):} $f(a) = f(b) \Rightarrow a = b$. \\
\textbf{Surjective (Onto):} $\forall y \in B, \exists x \in A, f(x) = y$. \\
\textbf{Bijective:} Both injective and surjective. \\
\textbf{Inverse:} $f^{-1}$ exists iff $f$ is a bijection. \\
\textbf{Composition:} $(f \circ g)(x) = f(g(x))$.

\section{Number Theory}
\textbf{Divisibility:} "Does $a$ divide $b$?" $\Rightarrow b=ak$. \\
\textbf{GCD/LCM:} "Find $\gcd$ and $\text{lcm}$" $\Rightarrow \gcd(a,b) \cdot \text{lcm}(a,b) = ab$. \\
\textbf{Modular Powers:} "Find $a^{X} \mod p$ (prime $p$)" $\Rightarrow a^{p-1} \equiv 1$ (\textbf{FLT}). \\
\textbf{Linear Combo:} "Write $\gcd$ as $sa+tb$" $\Rightarrow$ \textbf{B\'ezout's Thm}. \\
\textbf{Modular Inverse:} "Solve $ax \equiv 1 \pmod m$" $\Rightarrow$ Exists if $\gcd(a,m)=1$. \\
\textbf{Simult. Congruence:} "Find $x \equiv a \pmod m, x \equiv b \pmod n$" $\Rightarrow$ \textbf{CRT}. \\
\textbf{Primality Test:} "Is $p$ prime?" $\Rightarrow (p-1)! \equiv -1 \pmod p$ (\textbf{Wilson's}). \\
\textbf{Primes $\le x$:} "How many primes?" $\Rightarrow \pi(x) \approx x/\ln x$.

\section{Algorithms \& Complexity}
\textbf{Big-O:} $f(n) \le Cg(n)$ for $n > k$. \quad \textbf{Growth:} $1 < \log n < n < n \log n < n^2 < 2^n < n!$. \\
\textbf{Master Thm:} $T(n) = aT(n/b) + cn^d$: \\
1. $a < b^d \Rightarrow O(n^d)$ \quad 2. $a = b^d \Rightarrow O(n^d \log n)$ \quad 3. $a > b^d \Rightarrow O(n^{\log_b a})$ \\
\textbf{Search:} Linear $O(n)$, Binary $O(\log n)$. \quad \textbf{Sort:} Bubble/Inertion $O(n^2)$.

\section{Sequences \& Sums}
\textbf{Arithmetic:} $a_n = a + (n-1)d$. \textbf{Sum:} $S_n = \frac{n(2a+(n-1)d)}{2}$. \\
\textbf{Geometric:} $a_n = ar^{n-1}$. \textbf{Sum:} $S_n = \frac{a(r^n-1)}{r-1}$ ($r \neq 1$). \\
\textbf{Series:} $\sum_{i=1}^n i = \frac{n(n+1)}{2}$, $\sum i^2 = \frac{n(n+1)(2n+1)}{6}$.

\section{Counting}
\textbf{Sequential Tasks:} "Step 1 \underline{and} Step 2" $\Rightarrow n_1 \cdot n_2$ (\textbf{Product}). \\
\textbf{Disjoint Choices:} "Option 1 \underline{or} Option 2" $\Rightarrow n_1 + n_2$ (\textbf{Sum}). \\
\textbf{Overlapping Sets:} "Count $|A \cup B|$" $\Rightarrow |A|+|B|-|A \cap B|$ (\textbf{Subtract}). \\
\textbf{Overcounting:} "Ways if order doesn't matter yet" $\Rightarrow n/d$ (\textbf{Division}). \\
\textbf{Guaranteed Match:} "Min objects to ensure $k$ in bin" $\Rightarrow \lceil N/k \rceil$ (\textbf{PHP}). \\
\textbf{Mappings:} "Functions $A \to B$" $\Rightarrow |B|^{|A|}$; "Injective" $\Rightarrow P(|B|, |A|)$.

\section{Perms \& Combs}
\textbf{Arrangements:} "Order matters, no rep" $\Rightarrow P(n,r) = \frac{n!}{(n-r)!}$. \\
\textbf{Committees/Sets:} "Order ign, no rep" $\Rightarrow \binom{n}{r} = \frac{n!}{r!(n-r)!}$. \\
\textbf{Passwords:} "Order matters, w/ rep" $\Rightarrow n^r$. \\
\textbf{Donut/Soda choice:} "Order ign, w/ rep" $\Rightarrow \binom{n+r-1}{r}$ (\textbf{Stars/Bars}). \\
\textbf{Anagrams:} "Permute word with $n_i$ repeats" $\Rightarrow \frac{n!}{n_1!n_2!\dots n_k!}$. \\
\textbf{Dist. Objs $\to$ Dist. Boxes:} "Put $n$ items into $k$ boxes" $\Rightarrow \frac{n!}{n_1!n_2!\dots n_k!}$. \\
\textbf{Hockey-Stick:} "Sum $\binom{i}{r}$ for $i=r \to n$" $\Rightarrow \binom{n+1}{r+1}$. \\
\textbf{Vandermonde's:} "Pick $r$ from two groups ($m, n$)" $\Rightarrow \binom{m+n}{r} = \sum \binom{m}{r-k}\binom{n}{k}$. \\
\textbf{Derangement:} "Ways to fix 0 items in original spot" $\Rightarrow D_n$. \\
\textbf{Identities:} Total subsets $\Rightarrow 2^n$; Alternating sum $\Rightarrow 0$.

\section{Relations}
$R \subseteq A \times A$:
\begin{itemize}
    \item \textbf{Reflexive:} $\forall a, (a,a) \in R$
    \item \textbf{Symmetric:} $(a,b) \in R \Rightarrow (b,a) \in R$
    \item \textbf{Antisymmetric:} $(a,b),(b,a) \in R \Rightarrow a=b$
    \item \textbf{Transitive:} $(a,b),(b,c) \in R \Rightarrow (a,c) \in R$
\end{itemize}
\textbf{Equiv. Relation:} Refl, Symm, Trans. \\
\textbf{Partial Order (Poset):} Refl, Antisymm, Trans. \\
\textbf{\# Relations on $n$ elements:} $2^{n^2}$. \textbf{Reflexive:} $2^{n(n-1)}$.

\section{Recurrence Relations}
\textbf{Lin. Homog. order $k$:} $a_n = c_1 a_{n-1} + \dots + c_k a_{n-k}$. \\
\textbf{Char. Eq:} $r^k - c_1 r^{k-1} - \dots - c_k = 0$. \\
1. \textbf{Distinct roots $r_1, r_2$:} $a_n = \alpha_1 r_1^n + \alpha_2 r_2^n$. \\
2. \textbf{Repeated root $r_1$:} $a_n = (\alpha_1 + \alpha_2 n) r_1^n$. \\
\textbf{Solving:} 1. Find char eq. 2. Find roots. 3. Use IC to solve $\alpha_i$.

\section{Induction \& Probability}
\textbf{Goal:} $\forall n \ge a, P(n)$ \quad \textbf{Basis:} Show $P(a)$. \\
\textbf{Inductive:} Assume $P(k)$ (\textbf{I.H.}), show $P(k+1)$. \\
\textbf{Strong:} Assume $P(a) \land \dots \land P(k)$, show $P(k+1)$. \\
\textbf{Prob(Cause | Effect):} "Given result $B$, prob of origin $A$?" $\Rightarrow$ \textbf{Bayes}. \\
\textbf{Linearity of $E$:} "Average sum of indep events" $\Rightarrow E(X)+E(Y)$. \\
\textbf{Multiple Trials:} "Prob of $k$ successes in $n$ tries" $\Rightarrow$ \textbf{Bernoulli}. \\
\textbf{Average Search:} "Find avg steps for $n$ items" $\Rightarrow$ \textbf{Exp. Value}.

\section{Graph Theory}
\textbf{Handshaking:} "Find edges or degrees" $\Rightarrow \sum \deg(v) = 2e$. \\
\textbf{Planarity:} "Is it planar? $V, E, F$?" $\Rightarrow V-E+F=2$ (\textbf{Euler}). \\
\textbf{Bound:} "Is $K_{n,m}$ planar?" $\Rightarrow E \le 3V-6$ or $2V-4$. \\
\textbf{Connectivity:} "Is graph connected?" $\Rightarrow$ Exists path $\forall u,v$. \\
\textbf{Trees:} "Properties of trees?" $\Rightarrow E=V-1$, No cycles. \\
\textbf{Hamilton:} "Circuit using every \underline{vertex}?" $\Rightarrow$ Dirac/Ore. \\
\textbf{Eulerian:} "Circuit using every \underline{edge}?" $\Rightarrow$ All degrees even.
