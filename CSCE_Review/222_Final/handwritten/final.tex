\section{Counting}
$T_1 = n_1, T_2 = n_2$\\
\textbf{{Product Rule:}} $T_1 \land T_2 = n_1 \times n_2 \Rightarrow 7(T_i) = n_i^7$\\
\textbf{For Sets:}$|A_1 \times A_2 \times \ldots \times A_m | = |A_1| \times |A_2| \times \ldots \times |A_m|$\\
\textbf{Sum Rule:} $T_1 \lor T_2 = n_1 + n_2$\\
\textbf{Disjoint S:}$|A_1 \cup A_2 \cup ... \cup A_m | = |A_1| + |A_2| + ... + |A_m|$\\
\textbf{Subtraction Rule:} $T_1 \lor T_2 = n_1 + n_2 - (T_1 \cap T_2)$\\
\textbf{Non disjoint sets:} $A_1 \cup A_2 = |A_1| + |A_2| - |A_1 \cap A_2|$\\
\textbf{Division Rule:} $\text{Unique Ways} = \frac{n}{d}$\\
\textbf{For Sets:} $\text{If a set } A \text{ is split into } n \text{ equal groups of size } d,\\
n = \frac{|A|}{d}$\\
\textbf{Ex:} n seats at round table, Circular Ways = $\frac{n!}{n}$\\
\textbf{r-Permutations:} set of r ordered arrangements\\
\textbf{r-Combinations:} set of r unordered arrangements\\
P(n,r) = $\frac{n!}{(n-r)!}$\hspace{1cm} C(n,r) = $\binom{n}{r}= \frac{n!}{r!(n-r)!}$\\
P(n,r) with repetitions allowed = $n^r$\\
C(n,r) with repetitions (bins) allowed = $C(n-1+r,r)$\\
C order doesnt matter, therefore $C(n, r) = C(n, n-r)$\\

$\sum_{k=0}^{n}\binom{n}{k} = (1+1)^n = 2^n$\\
$\sum_{k=0}^{n}(-1)^k\binom{n}{k}= (-1+1)^n=0$

\textbf{Linear Homogenous recurrence relation:} $a_n = c_1a_{n-1} + c_2a_{n-2}+...+c_ka_{n-k}$\\
\textbf{Linear:} RHS is a \underbar{sum} of previous terms$^1$ multiplied by some coeficient\\
\textbf{Homogeneous:} all RHS terms are multiples of previous terms$_{n-i}$\\
\textbf{Constant Coefficients:} all coefficients are constant, $\neq 0$\\
\textbf{Degree:} The number of previous terms in sequence\\
Degree determines NO. initial conditions $(a_0 = C_0),...,(a_{k-1}=C_{k-1})$\\
\textbf{Characteristic Equations and Roots:}\\
$a_n = c_1a_{n-1} + c_2a_{n-2}+...+c_ka_{n-k}$\hspace{0.2cm} , \hspace{0.2cm}$a_n = r^n$\hspace{0.2cm}$\Rightarrow$\hspace{0.2cm}\\
Where the solutions are the roots: $r^k - c_1r^{k-1}-...-c_{k-1}r-c_k=0$\\
Solution = $(r_1,...,r_k)^T \to a_n = \alpha_1r_1^n + \alpha_2r_2^n$, $\alpha$ solved using initial condition\\

f(n) = a \cdot f(n/b) + c \cdot n^d \implies

\begin{aligned}

\end{aligned}
\end{equation*}

\section{Relations}
$R \subseteq A \times B \implies (aRb \iff (a, b) \in R)$, not all (a,b) are in R\\
\begin{itemize}
    \item \textbf{Reflexive:} $\forall a \in A, aRa$
    \item \textbf{Symmetric:} $aRb \implies bRa$
    \item \textbf{Antisymmetric:} $(aRb \land bRa) \implies a = b$\hspace{0.3cm} \textit{if both R work, must be reflexive}
    \item \textbf{Transitive:} $(aRb \land bRc) \implies aRc$
    \item {Vacuously True:} $P \implies Q$ is true if $P$ is always false. \textit{(premise never met)}
\end{itemize}
you may use substitution to prove a=b or contradict\\
\textbf{Composition ($S \circ R$):} $\{(a, c) \mid \exists b \in B, (a, b) \in R \land (b, c) \in S \}$\\
Using $S \circ R$ if a is parent of b, b is parent of c $\to$ a is grandparent of c in $S \circ R$\\
R on set A is an \textbf{equivalence} (like row equiv) if reflexive, symmetric, transitive:\\
$(a,b) \in R \to a\sim b$\\
\begin{itemize}
    \item \textbf{Powers ($R^n$):} $R^1 = R, R^{n+1} = R^n \circ R$ \hfill \textit{(path of length $n$)}
    \item \textbf{Equivalence Relation ($\sim$):} A relation that is Refl. + Symm. + Trans.
    \item \textbf{Equivalence Class $[x]$:} $\{y \in S \mid x \sim y\}$ \hfill \textit{(The set that act like x)}
    \item \textbf{Row Equivalence ($A \sim B$):} An equivalence relation where $A, B$ share the same RREF.
\end{itemize}

\textbf{Proof:}

\begin{itemize}
    \item \textbf{Basis Step:} We show that $P(a)$ holds.

    \item \textbf{Inductive Step:} Assume the inductive hypothesis that $P(k)$ is true for an arbitrary integer $k\geq a$. That is, [State $P(k)$].\\
    We must show that $P(k+1)$ is true [State $P(k+1)$]. 
    \begin{align*}
    \text{RHS}(k+1) &= \underbrace{\text{General Terms up to } k}_{RHS(k)} + \text{Term}(k+1) \\
    &= \text{RHS}(k) + \text{Term}(k+1) \\
    &= \dots \text{ [Algebraic Simplification] } \dots \\

\end{align*}
$\therefore P(k) \to P(k+1) holds$\\

\end{itemize}

\textbf{Proof:}
\begin{itemize}
    \item \textbf{Basis Step:} For $n = [n_0]$, we have:
    \[ [LHS(n_0)] = [\text{value}] \quad \text{and} \quad [RHS(n_0)] = [\text{value}] \]
    Since $[\text{value}] < [\text{value}]$, $P([n_0])$ is true.

    \item \textbf{Inductive Step:} Assume $P(k)$ is true for an arbitrary integer $k \geq [n_0]$. 
    That is, assume:
    \[ [LHS(k)] < [RHS(k)] \quad \text{(Inductive Hypothesis)} \]
    
    We must show that $P(k+1)$ is true: $[LHS(k+1)] < [RHS(k+1)]$.
    
    Starting with the left-hand side:
    \begin{align*}
        [LHS(k+1)] &= [\text{Manipulate to find } LHS(k)] \\

    \end{align*}
\end{itemize}
